Гусь.Веб открывает файлы с диска: напишите страницу в Блокноте или IDE, сохраните как /home/ИМЯ/index.html и откройте её адресом file:///home/ИМЯ/index.html (или Файл → Открыть файл). F5 перезагружает страницу после правок.

Каркас страницы

<!DOCTYPE html>
<html lang="ru">
<head>
  <meta charset="utf-8">
  <title>Моя страница</title>
  <style>
    body { font-family: sans-serif; margin: 24px; background: #f4f6fa; }
    .card { background: #fff; border-radius: 10px; padding: 12px; }
  </style>
</head>
<body>
  <div class="card"><h1>Привет!</h1><button id="b">Нажми</button></div>
  <script>
    document.getElementById('b').addEventListener('click', e => e.target.textContent = 'Га-га!');
  </script>
</body>
</html>

Canvas: игровой цикл

<canvas id="c" width="320" height="200"></canvas>
<script>
const ctx = document.getElementById('c').getContext('2d');
let x = 0;
const keys = {};
addEventListener('keydown', e => keys[e.key] = true);
addEventListener('keyup', e => keys[e.key] = false);
function frame() {
  if (keys.ArrowRight) x += 2;
  if (keys.ArrowLeft) x -= 2;
  ctx.fillStyle = '#123'; ctx.fillRect(0, 0, 320, 200);
  ctx.fillStyle = '#fc0'; ctx.fillRect(x, 90, 20, 20);
  requestAnimationFrame(frame);
}
frame();
</script>
