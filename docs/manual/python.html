Python — пакет python: BarnStore или sudo brm -S python. Программы пишутся в Barnix Code (F5 — отладчик с точками останова, Ctrl+F5 — простой запуск; вывод и ошибки — в нижней панели; подсказки по Ctrl+Space) или в любом редакторе.

python3 hello.py        # запустить программу
python3                 # интерактивный режим, выход: exit()
python3 -c "print(2**100)"

Первая программа

name = input("Как тебя зовут? ")
print(f"Привет, {name}!")
for i in range(1, 6):
    print(i, "в квадрате =", i * i)

Основы

numbers = [5, 3, 8, 1]
numbers.sort()
squares = [n * n for n in numbers]
age = {"Гусь": 3, "Утка": 2}
for name, years in age.items():
    print(name, years)

def fib(n):
    return n if n < 2 else fib(n - 1) + fib(n - 2)

class Goose:
    def __init__(self, name):
        self.name = name
    def honk(self):
        return f"{self.name}: Га-га!"

Файлы

with open("notes.txt", "w") as f:
    f.write("строка\n")
with open("notes.txt") as f:
    for line in f:
        print(line.strip())

Обычный пользователь пишет в свою папку и в /tmp.
